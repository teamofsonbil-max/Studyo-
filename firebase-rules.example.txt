// REVIEW THIS EXAMPLE BEFORE USING. Adapt rules to the app's requirements.
// Firestore Rules are enforced on Firebase, not by the Android app.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    match /people/{userId} {
      allow read: if signedIn();
      allow create, update: if signedIn() && request.auth.uid == userId;
    }
    match /posts/{postId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow update: if signedIn(); // Tighten likes/comments updates with validation before production.
      allow delete: if signedIn() && resource.data.uid == request.auth.uid;
    }
    match /follows/{followId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.a == request.auth.uid;
      allow delete: if signedIn() && resource.data.a == request.auth.uid;
    }
    match /dm/{messageId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.from == request.auth.uid;
    }
    match /stories/{storyId} {
      allow read: if signedIn();
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if signedIn() && resource.data.uid == request.auth.uid;
    }
  }
}
